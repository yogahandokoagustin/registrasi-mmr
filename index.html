async function loadBlokTanpaRegistrasi() { try { const response = await callApi('getBlokTanpaRegistrasi'); // PENJAGAAN: Pastikan response dari server benar-benar berupa Array. // Jika bukan Array, berarti API bermasalah atau belum di-deploy ulang. if (!Array.isArray(response)) { const errorMsg = (response && response.message) ? response.message : 'Format data dari server tidak valid. Pastikan Apps Script sudah di-deploy ulang (New Version)!'; throw new Error(errorMsg); } const list = response; const body = document.getElementById('belumTableBody'); body.innerHTML = ''; // Jika list.length > 0, sembunyikan pesan kosong. Jika 0, tampilkan pesan kosong. document.getElementById('belumEmpty').classList.toggle('hidden', list.length > 0); list.forEach(function(r) { const tr = document.createElement('tr'); tr.innerHTML = ''+r.no+''+escapeHtml(r.blok)+''+escapeHtml(r.nama)+''; body.appendChild(tr); }); } catch (err) { // Menampilkan pesan error yang jelas di layar jika terjadi masalah showMsg(err.message || 'Gagal memuat data', true); } }
